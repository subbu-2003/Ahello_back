import {
  Input, Select, DatePicker, TimePicker,

} from "antd";
import {
  EnvironmentOutlined, QrcodeOutlined,
  FileImageOutlined, PaperClipOutlined,
} from "@ant-design/icons";
import { QRCodeCanvas } from "qrcode.react";
import LocationPickerField from "../../HostServices/AllForms/LocationPickerField";

// ─── Signature Pad ────────────────────────────────────────────────────────────
export const SignaturePad = ({ value, onChange, height = 140 }) => {
  const canvasRef  = useRef(null);
  const wrapperRef = useRef(null);
  const drawing    = useRef(false);

  useEffect(() => {
    const canvas  = canvasRef.current;
    const wrapper = wrapperRef.current;
    if (!canvas || !wrapper) return;
    const ratio       = window.devicePixelRatio || 1;
    canvas.width      = wrapper.offsetWidth * ratio;
    canvas.height     = height * ratio;
    canvas.style.width  = `${wrapper.offsetWidth}px`;
    canvas.style.height = `${height}px`;
    const ctx = canvas.getContext("2d");
    ctx.scale(ratio, ratio);
    ctx.lineWidth   = 2;
    ctx.lineCap     = "round";
    ctx.strokeStyle = "#000000";
    if (value) {
      const img = new Image();
      img.src = value;
      img.onload = () => ctx.drawImage(img, 0, 0, wrapper.offsetWidth, height);
    }
  }, [height]);

  const getPos = (e) => {
    const rect  = canvasRef.current.getBoundingClientRect();
    const point = e.touches ? e.touches[0] : e;
    return { x: point.clientX - rect.left, y: point.clientY - rect.top };
  };
  const startDrawing = (e) => {
    drawing.current = true;
    const ctx = canvasRef.current.getContext("2d");
    const pos = getPos(e);
    ctx.beginPath(); ctx.moveTo(pos.x, pos.y);
  };
  const draw = (e) => {
    if (!drawing.current) return;
    e.preventDefault();
    const ctx = canvasRef.current.getContext("2d");
    const pos = getPos(e);
    ctx.lineTo(pos.x, pos.y); ctx.stroke();
  };
  const stopDrawing = () => {
    if (!drawing.current) return;
    drawing.current = false;
    onChange?.(canvasRef.current.toDataURL("image/png"));
  };
  const clearSignature = () => {
    const canvas = canvasRef.current;
    canvas.getContext("2d").clearRect(0, 0, canvas.width, canvas.height);
    onChange?.("");
  };

  return (
    <div ref={wrapperRef} style={{ border: "1px dashed #ccc", borderRadius: 8, overflow: "hidden", position: "relative", background: "#fff" }}>
      <canvas
        ref={canvasRef}
        style={{ width: "100%", height, display: "block", cursor: "crosshair", touchAction: "none" }}
        onMouseDown={startDrawing} onMouseMove={draw} onMouseUp={stopDrawing} onMouseLeave={stopDrawing}
        onTouchStart={startDrawing} onTouchMove={draw} onTouchEnd={stopDrawing}
      />
      <Button size="small" onClick={clearSignature} style={{ position: "absolute", right: 10, bottom: 10, fontSize: 11 }}>Clear</Button>
      <span style={{ position: "absolute", left: 10, bottom: 12, fontSize: 11, color: "#999", pointerEvents: "none" }}>Draw signature here</span>
    </div>
  );
};

// ─── OTP Input ────────────────────────────────────────────────────────────────
export const OtpInput = ({ digits = 6 }) => (
  <div style={{ display: "flex", gap: 8 }}>
    {Array.from({ length: digits }).map((_, i) => (
      <Input key={i} maxLength={1} style={{ width: 40, textAlign: "center", fontWeight: 600, fontSize: 18, padding: "4px 0", borderRadius: 6 }} />
    ))}
  </div>
);

// ─── QR Scanner placeholder ───────────────────────────────────────────────────
export const QrScannerPlaceholder = ({ link }) => {
  if (!link) return (
    <div style={{ border: "2px dashed #d9d9d9", borderRadius: 10, padding: "40px 20px", textAlign: "center", background: "#fafafa" }}>
      <QrcodeOutlined style={{ fontSize: 50, color: "#bbb", marginBottom: 10 }} />
      <div style={{ color: "#999", fontSize: 13 }}>No QR link added</div>
    </div>
  );
  return (
    <div style={{ display: "flex", justifyContent: "center", padding: 12, background: "#fff", borderRadius: 12, border: "1px solid #e5e5e5" }}>
      <QRCodeCanvas id="dynamic-qr-preview" value={link} size={180} includeMargin level="H" />
    </div>
  );
};

// ─── Universal field renderer ─────────────────────────────────────────────────
export const renderFieldInput = (field, onFieldChange) => {
  const { type, placeholder, options = [] } = field;
  const ddOpts  = options.map((o) => ({ value: o, label: o }));
  const hasOpts = options.length > 0;

  switch (type) {
    case "short_text":      return <Input placeholder={placeholder} />;
    case "number":          return <InputNumber placeholder={placeholder} style={{ width: "100%" }} />;
    case "email":           return <Input type="email" placeholder={placeholder} />;
    case "phone":           return <Input type="tel" placeholder={placeholder} addonBefore="+91" />;
    case "url":             return <Input type="url" placeholder={placeholder} addonBefore="https://" />;
    case "textarea":        return <Input.TextArea rows={3} placeholder={placeholder} style={{ borderRadius: 6 }} />;
    case "multiline_text":  return <Input.TextArea rows={4} placeholder={placeholder} style={{ borderRadius: 6 }} />;
    case "pan":             return <Input placeholder={placeholder} maxLength={10} style={{ textTransform: "uppercase", letterSpacing: 2 }} />;
    case "pincode":         return <Input placeholder={placeholder} maxLength={6} />;
    case "price":           return <InputNumber placeholder={placeholder} style={{ width: "100%" }} prefix="₹" min={0} precision={2} />;
    case "dropdown":
      return <Select placeholder={placeholder} style={{ width: "100%" }} options={hasOpts ? ddOpts : [{ value: "_", label: "No options yet" }]} />;
    case "multi_select":
      return <Select mode="multiple" placeholder={placeholder} style={{ width: "100%" }} options={hasOpts ? ddOpts : [{ value: "_", label: "No options yet" }]} />;
    case "checkbox":
      return (
        <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
          {hasOpts ? options.map((o, i) => <Checkbox key={i}>{o}</Checkbox>) : <Checkbox>{field.label}</Checkbox>}
        </div>
      );
    case "radio":
      return (
        <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
          {hasOpts ? options.map((o, i) => <Radio key={i}>{o}</Radio>) : <Radio>{field.label}</Radio>}
        </div>
      );
    case "date":            return <DatePicker placeholder={placeholder} style={{ width: "100%" }} />;
    case "time_picker":     return <TimePicker placeholder={placeholder} style={{ width: "100%" }} use12Hours format="h:mm a" />;
    case "datetime_picker": return <DatePicker showTime placeholder={placeholder} style={{ width: "100%" }} />;
    case "toggle_switch":
      return (
        <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
          <Switch /><span style={{ fontSize: 13, color: "#555" }}>Toggle on / off</span>
        </div>
      );
    case "rating_input":    return <Rate />;
    case "signature_pad":
      return <SignaturePad value={field.signature} onChange={(val) => { field.signature = val; }} />;
    case "otp":             return <OtpInput digits={6} />;
    case "qr_scanner":      return <QrScannerPlaceholder link={field.qrLink} />;
    case "location_picker":
      return <LocationPickerField value={field.location} onChange={(loc) => onFieldChange?.({ ...field, location: loc })} />;
    case "consent_checkbox":
      return <Checkbox><span style={{ fontSize: 13 }}>{placeholder || "I agree to the terms and conditions"}</span></Checkbox>;
    case "image_upload":
  return (
    <Upload accept="image/*" showUploadList={false} beforeUpload={() => false}>
          <Button
              icon={<FileImageOutlined />}
              style={{ borderRadius: 8, borderStyle: "dashed", width: "100%", height: 36, color: "#6b7280", fontSize: 12 }}
          >
              Click to upload image
          </Button>
          <div style={{ fontSize: 11, color: "#9ca3af", marginTop: 4, textAlign: "center" }}>
              PNG, JPG, WEBP
          </div>
    </Upload>
  );

case "file_upload":
  return (
    <Upload accept=".pdf,.doc,.docx,.xls,.xlsx,.txt,.csv,.zip" showUploadList={false} beforeUpload={() => false}>
          <Button
              icon={<PaperClipOutlined />}
              style={{ borderRadius: 8, borderStyle: "dashed", width: "100%", height: 36, color: "#6b7280", fontSize: 12 }}
          >
              Click to upload file
          </Button>
          <div style={{ fontSize: 11, color: "#9ca3af", marginTop: 4, textAlign: "center" }}>
              PDF, DOC, XLS, CSV
          </div>
    </Upload>
  );
    default:
      return <Input placeholder={placeholder} />;
  }
};
